% forward row
\node[input] (x)  at (0,0)    {input\\$\mathbf{x} \in \mathbb{R}^{n}$};
\node[op]    (a1) at (3.2,0)  {$\mathbf{a} = \mathbf{W}_1 \mathbf{x} + \mathbf{b}_1$\\$\mathbb{R}^{d}$};
\node[op]    (h)  at (6.4,0)  {$\mathbf{h} = \sigma(\mathbf{a})$\\(element-wise)};
\node[op]    (z)  at (9.6,0)  {$\mathbf{z} = \mathbf{W}_2 \mathbf{h} + \mathbf{b}_2$\\logits, $\mathbb{R}^{3}$};

\node[gbox] (gz) at (12.8,-2.4) {$\dfrac{\partial \ell}{\partial \mathbf{z}} = \hat{\mathbf{p}} - \mathbf{e}_{y}$};

\draw[back] (gz) -- (gh); \draw[back] (gh) -- (ga);
% gradients with respect to the weights
\node[gbox] (gw2) at (9.6,-4.6)  {$\dfrac{\partial \ell}{\partial \mathbf{W}_2} = \dfrac{\partial \ell}{\partial \mathbf{z}}\, \mathbf{h}^{\top}$};
\node[gbox] (gw1) at (3.2,-4.6)  {$\dfrac{\partial \ell}{\partial \mathbf{W}_1} = \dfrac{\partial \ell}{\partial \mathbf{a}}\, \mathbf{x}^{\top}$};

\draw[back] (ga.south) |- (gw1.east);
